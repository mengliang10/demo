Events.on(engine, 'beforeUpdate', () => {
  bodies.forEach((b1, i) => {
    bodies.slice(i+1).forEach(b2 => {
      const d = Vector.sub(b2.position, b1.position);
      const f = (G * b1.mass * b2.mass) / (Vector.magnitude(d)**2 + eps);
      Body.applyForce(b1, b1.position, Vector.mult(Vector.normalise(d), f));
    });

});
